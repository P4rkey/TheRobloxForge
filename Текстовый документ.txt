29function InteractiveBackground() {
30  const canvasRef = useRef<HTMLCanvasElement | null>(null);
31
32  useEffect(() => {
33    const canvas = canvasRef.current;
34    if (!canvas) return;
35    const ctx = canvas.getContext("2d");
36    if (!ctx) return;
37
38    let animationFrameId: number;
39    let width = (canvas.width = window.innerWidth);
40    let height = (canvas.height = window.innerHeight);
41
42    const handleResize = () => {
43      if (!canvas) return;
44      width = canvas.width = window.innerWidth;
45      height = canvas.height = window.innerHeight;
46    };
47    window.addEventListener("resize", handleResize);
48
49    const particleCount = 55;
50    const particles: Array<{
51      x: number;
52      y: number;
53      vx: number;
54      vy: number;
55      size: number;
56      color: string;
57      alpha: number;
58      pulseSpeed: number;
59    }> = [];
60
61    const colors = ["#8B5CF6", "#EC4899", "#F43F5E", "#D946EF", "#F59E0B", "#38BDF8"];
62
63    for (let i = 0; i < particleCount; i++) {
64      particles.push({
65        x: Math.random() * width,
66        y: Math.random() * height,
67        vx: (Math.random() - 0.5) * 1.2,
68        vy: (Math.random() - 0.5) * 1.2,
69        size: Math.random() * 3.5 + 1.5,
70        color: colors[Math.floor(Math.random() * colors.length)],
71        alpha: Math.random() * 0.7 + 0.3,
72        pulseSpeed: Math.random() * 0.03 + 0.01,
73      });
74    }
75
76    let mouseX = width / 2;
77    let mouseY = height / 2;
78
79    const handleMouseMove = (e: MouseEvent) => {
80      mouseX = e.clientX;
81      mouseY = e.clientY;
82    };
83    window.addEventListener("mousemove", handleMouseMove);
84
85    let time = 0;
86    const render = () => {
87      time += 0.02;
88      ctx.clearRect(0, 0, width, height);
89
90      for (let i = 0; i < particles.length; i++) {
91        for (let j = i + 1; j < particles.length; j++) {
92          const dx = particles[i].x - particles[j].x;
93          const dy = particles[i].y - particles[j].y;
94          const dist = Math.sqrt(dx * dx + dy * dy);
95
96          if (dist < 140) {
97            ctx.beginPath();
98            ctx.moveTo(particles[i].x, particles[i].y);
99            ctx.lineTo(particles[j].x, particles[j].y);
100            ctx.strokeStyle = `rgba(217, 70, 239, ${0.25 * (1 - dist / 140)})`;
101            ctx.lineWidth = 1.2;
102            ctx.stroke();
103          }
104        }
105      }
106
107      particles.forEach((p, idx) => {
108        p.x += p.vx + Math.sin(time * 0.5 + idx) * 0.3;
109        p.y += p.vy + Math.cos(time * 0.5 + idx) * 0.3;
110
111        if (p.x < 0) p.x = width;
112        if (p.x > width) p.x = 0;
113        if (p.y < 0) p.y = height;
114        if (p.y > height) p.y = 0;
115
116        const mdx = mouseX - p.x;
117        const mdy = mouseY - p.y;
118        const mdist = Math.sqrt(mdx * mdx + mdy * mdy);
119        if (mdist < 160) {
120          p.x -= (mdx / mdist) * 0.8;
121          p.y -= (mdy / mdist) * 0.8;
122        }
123
124        const currentAlpha = p.alpha + Math.sin(time * p.pulseSpeed * 10) * 0.2;
125
126        ctx.beginPath();
127        ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
128        ctx.fillStyle = p.color;
129        ctx.globalAlpha = Math.max(0.1, Math.min(1, currentAlpha));
130        ctx.shadowBlur = 14;
131        ctx.shadowColor = p.color;
132        ctx.fill();
133        ctx.shadowBlur = 0;
134        ctx.globalAlpha = 1.0;
135      });
136
137      animationFrameId = requestAnimationFrame(render);
138    };
139
140    render();
141
142    return () => {
143      window.removeEventListener("resize", handleResize);
144      window.removeEventListener("mousemove", handleMouseMove);
145      cancelAnimationFrame(animationFrameId);
146    };
147  }, []);
148
149  return (
150    <canvas
151      ref={canvasRef}
152      className="fixed inset-0 pointer-events-none z-0 opacity-85"
153    />
154  );
155}
